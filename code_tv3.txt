package com.skribbl.common.protocol;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.SerializationFeature;

import java.io.IOException;

/**
 * High-performance, thread-safe JSON Codec for NDJSON (Newline-Delimited JSON) protocol.
 * Ensures every serialized message is on a single line without formatting breaks.
 */
public final class JsonCodec {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    static {
        MAPPER.configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);
        MAPPER.configure(SerializationFeature.INDENT_OUTPUT, false);
    }

    private JsonCodec() {
        // Utility class
    }

    /**
     * @return Underlying configured ObjectMapper instance
     */
    public static ObjectMapper getMapper() {
        return MAPPER;
    }

    /**
     * Serializes a NetworkMessage into a single-line JSON string.
     *
     * @param message Message to serialize
     * @return Single line JSON string
     * @throws JsonProcessingException if serialization fails
     */
    public static String serialize(NetworkMessage message) throws JsonProcessingException {
        return MAPPER.writeValueAsString(message);
    }

    /**
     * Convenience method to wrap type and payload into a NetworkMessage and serialize to JSON.
     *
     * @param type MessageType
     * @param payload Object payload (can be null)
     * @return Single line JSON string
     * @throws JsonProcessingException if serialization fails
     */
    public static String serializeMessage(MessageType type, Object payload) throws JsonProcessingException {
        JsonNode node = payload != null ? MAPPER.valueToTree(payload) : null;
        NetworkMessage msg = NetworkMessage.of(type, node);
        return serialize(msg);
    }

    /**
     * Deserializes a single-line JSON string into a NetworkMessage envelope.
     *
     * @param jsonLine Line of JSON received from socket
     * @return Parsed NetworkMessage
     * @throws IOException if parsing fails or invalid JSON
     */
    public static NetworkMessage deserialize(String jsonLine) throws IOException {
        if (jsonLine == null || jsonLine.isBlank()) {
            throw new IllegalArgumentException("JSON line cannot be null or blank");
        }
        return MAPPER.readValue(jsonLine, NetworkMessage.class);
    }

    /**
     * Decodes the payload of a NetworkMessage into a concrete target class (e.g. record/DTO).
     *
     * @param message NetworkMessage containing the payload
     * @param clazz Target class
     * @param <T> Target type
     * @return Decoded payload object
     */
    public static <T> T decodePayload(NetworkMessage message, Class<T> clazz) {
        if (message == null || message.payload() == null || message.payload().isNull()) {
            return null;
        }
        return MAPPER.convertValue(message.payload(), clazz);
    }

    /**
     * Decodes the payload into a generic type via TypeReference.
     */
    public static <T> T decodePayload(NetworkMessage message, TypeReference<T> typeRef) {
        if (message == null || message.payload() == null || message.payload().isNull()) {
            return null;
        }
        return MAPPER.convertValue(message.payload(), typeRef);
    }

    /**
     * Converts any Java object to a Jackson JsonNode tree.
     */
    public static JsonNode toNode(Object object) {
        if (object == null) {
            return null;
        }
        return MAPPER.valueToTree(object);
    }
}

package com.skribbl.common.protocol;

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.node.NullNode;

/**
 * Standard protocol envelope wrapping every NDJSON message exchanged over TCP.
 * Formatted as a single JSON line terminated by newline (\n).
 *
 * @param type Message identifier matching {@link MessageType}
 * @param timestamp Epoch millisecond timestamp when the message was created
 * @param payload JSON data specific to the message type
 */
public record NetworkMessage(
        @JsonProperty("type") MessageType type,
        @JsonProperty("timestamp") long timestamp,
        @JsonProperty("payload") JsonNode payload
) {
    @JsonCreator
    public NetworkMessage {
        if (payload == null) {
            payload = NullNode.getInstance();
        }
    }

    public static NetworkMessage of(MessageType type, JsonNode payload) {
        return new NetworkMessage(type, System.currentTimeMillis(), payload);
    }

    public static NetworkMessage of(MessageType type) {
        return new NetworkMessage(type, System.currentTimeMillis(), NullNode.getInstance());
    }
}

    @Test
    @DisplayName("Should serialize NetworkMessage to single-line JSON without line breaks")
    void testSerializationSingleLine() throws IOException {
        ConnectReqDto req = new ConnectReqDto("PlayerOne");
        String json = JsonCodec.serializeMessage(MessageType.CONNECT_REQ, req);

        assertNotNull(json);
        assertFalse(json.contains("\n"), "Serialized JSON must not contain newlines");
        assertFalse(json.contains("\r"), "Serialized JSON must not contain carriage returns");
        assertTrue(json.contains("CONNECT_REQ"));
        assertTrue(json.contains("PlayerOne"));
    }

    @Test
    @DisplayName("Should deserialize JSON line and decode payload cleanly")
    void testDeserializationAndPayloadDecoding() throws IOException {
        DrawStrokeDto stroke = new DrawStrokeDto(10.5, 20.0, 35.2, 40.8, "#FF0000", 4, false);
        String json = JsonCodec.serializeMessage(MessageType.DRAW_STROKE, stroke);

        NetworkMessage msg = JsonCodec.deserialize(json);
        assertEquals(MessageType.DRAW_STROKE, msg.type());
        assertTrue(msg.timestamp() > 0);

        DrawStrokeDto decoded = JsonCodec.decodePayload(msg, DrawStrokeDto.class);
        assertNotNull(decoded);
        assertEquals(10.5, decoded.x1(), 0.001);
        assertEquals("#FF0000", decoded.color());
        assertEquals(4, decoded.thickness());
        assertFalse(decoded.eraser());
    }

    @DisplayName("Should serialize and deserialize complex RoomUpdateDto with nested player list")
    void testComplexPayload() throws IOException {
        PlayerDto p1 = new PlayerDto("p1", "Alice", 150, true, false, false);
        PlayerDto p2 = new PlayerDto("p2", "Bob", 200, false, true, true);

    }

    public void connectToServer(String host, int port) throws IOException {

        socketClient.send(MessageType.CONNECT_REQ, new ConnectReqDto(nickname));
    }

    public void sendDrawStroke(DrawStrokeDto dto) {
        if (!canDraw) return;
        socketClient.send(MessageType.DRAW_STROKE, dto);
    }

    public void sendClearCanvas() {
        if (!canDraw) return;
        socketClient.send(MessageType.CLEAR_CANVAS, null);
    }

        socketClient.send(MessageType.CHAT_SEND, new ChatSendDto(text.trim()));
    }

    /**
     * Nháº­n NetworkMessage tá»« ServerReaderThread vÃ  Ä‘áº©y cáº­p nháº­t UI vÃ o Platform.runLater().
     */
    public void handleIncomingMessage(NetworkMessage msg) {
        if (msg == null || msg.type() == null) return;

        switch (msg.type()) {
            case CONNECT_ACK -> {
                this.playerId = msg.payload().path("playerId").asText(nickname);
                Platform.runLater(() -> {
                    if (gameView != null) {
                        gameView.appendChatMessage("[Há»‡ thá»‘ng] Káº¿t ná»‘i thÃ nh cÃ´ng (ID: " + playerId + ")");
                    }
                });
            }
            case DRAW_STROKE -> {
                DrawStrokeDto stroke = JsonCodec.decodePayload(msg, DrawStrokeDto.class);
                if (stroke != null) {
                    Platform.runLater(() -> {
                        if (gameView != null) gameView.getDrawingCanvas().renderRemoteStroke(stroke);
                    });
                }
            }
            case CANVAS_SYNC -> {
                CanvasSyncDto syncDto = JsonCodec.decodePayload(msg, CanvasSyncDto.class);
                if (syncDto != null) {
                    Platform.runLater(() -> {
                        if (gameView != null) gameView.getDrawingCanvas().renderSyncHistory(syncDto.strokes());
                    });
                }
            }
            case CLEAR_CANVAS -> Platform.runLater(() -> {
                if (gameView != null) gameView.getDrawingCanvas().clearCanvasLocal();
            });
            case WORD_CHOICES -> {
                WordChoicesDto choices = JsonCodec.decodePayload(msg, WordChoicesDto.class);
                if (choices != null) {
                    Platform.runLater(() -> showWordChoicePopup(choices));

/**
 * Quáº£n lÃ½ káº¿t ná»‘i TCP Socket phÃ­a Client (UTF-8 NDJSON, báº­t TCP_NODELAY giáº£m Ä‘á»™ trá»… nÃ©t váº½).
 */
public class SocketClient {
    private Socket socket;
    private BufferedWriter writer;
    private BufferedReader reader;
    private ServerReaderThread readerThread;
    private final Object writeLock = new Object();

    public void connect(String host, int port, ClientController controller) throws IOException {
        socket = new Socket(host, port);
        socket.setTcpNoDelay(true);

        writer = new BufferedWriter(new OutputStreamWriter(socket.getOutputStream(), StandardCharsets.UTF_8));
        reader = new BufferedReader(new InputStreamReader(socket.getInputStream(), StandardCharsets.UTF_8));

        readerThread = new ServerReaderThread(reader, controller);
        readerThread.start();
    }

    public void send(MessageType type, Object payload) {
        if (writer == null || socket == null || socket.isClosed()) return;
        synchronized (writeLock) {
            try {
                String ndjsonLine = JsonCodec.serializeMessage(type, payload);
                writer.write(ndjsonLine);
                writer.newLine();
                writer.flush();
            } catch (IOException e) {
                System.err.println("[SocketClient] Gá»­i gÃ³i tin tháº¥t báº¡i: " + e.getMessage());
            }
        }
    }

    public void disconnect() {
        if (readerThread != null) {
            readerThread.stopReading();
        }
        try {
            if (socket != null && !socket.isClosed()) {
                socket.close();
            }
        } catch (IOException ignored) {
        }
    }

 */
public class StrokeThrottler {
    private static final double MIN_DISTANCE = 2.0;
    private double lastX = -1;
    private double lastY = -1;
    private boolean hasLastPoint = false;

    public boolean shouldSend(double x, double y) {
        if (!hasLastPoint) {
            update(x, y);
            return true;
        }
        double distance = Math.hypot(x - lastX, y - lastY);
        if (distance > MIN_DISTANCE) {
            update(x, y);
            return true;
        }
        return false;
    }

    public void reset() {
        hasLastPoint = false;
        lastX = -1;
        lastY = -1;
    }

    private void update(double x, double y) {
        this.lastX = x;
        this.lastY = y;
        this.hasLastPoint = true;
    }
